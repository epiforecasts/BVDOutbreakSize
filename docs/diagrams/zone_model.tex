% The health-zone model and how the joint fit enters it. A banner at the
% top gives the quantity sampled from the joint fit. Below it the zone
% model runs top to bottom on the left. Each purple box on the right is an
% input from the joint fit, level with the step it feeds and badged by how
% it enters: sampled through the meld, fixed at a posterior mean, or used
% to fit a prior. Purple symbols inside the zone boxes come from the joint
% fit.

\newcommand{\J}[1]{{\color{jointline}#1}}

\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm]

\def\xs{-2.6}     % zone stack centre
\def\ws{9.6cm}    % zone stack text width
\def\xj{5.15}      % joint input column centre
\def\wj{3.85cm}    % joint input text width

% ---- Introduction -------------------------------------------------------------
\node[note, text width=14.8cm, anchor=north west] (intro) at (-7.6,0)
  {\normalsize\textbf{Health-zone model}\\[3pt]
   \small Each patch's infections $\J{I_{p,t}}$ are sampled from the joint fit.
   The zone model only splits them over the patch's zones.
   Arrivals from other patches are placed by gravity pull, and the rest by
   each zone's force of infection.};

% ---- Sampled (melded) banner ---------------------------------------------------
\node[joint, text width=14.45cm, anchor=north] (meld) at (-0.08,-1.75)
  {\textbf{Sampled from the joint fit (melded)}: a normal fitted to the joint
   draws of log weekly patch infections and, when zones mix, log origin
   intensities at the cut-off\\[4pt]
   $\eta \sim \mathrm{Normal}(0, I)$, \quad $\zeta = L\eta$, \quad
   $I_{p,t} = \bar I_{p,t} \exp \zeta_p(t)$, \quad
   $\varepsilon_q = \bar\varepsilon_q \exp \zeta^{\varepsilon}_q$\\[4pt]
   {\footnotesize $\zeta$ meld deviation, $L$ Cholesky factor of the draws'
   covariance, $\zeta_p(t)$ interpolated daily from patch $p$'s weekly cells,
   $\zeta^{\varepsilon}_q$ origin $q$'s cell, $\varepsilon_q$ intensity of origin patch $q$.}\\[2pt]
   {\footnotesize \badge{mean} $\bar I_{p,t}$ and $\bar\varepsilon_q$ are the exp of
   the mean log draws.}};

% ---- Zone stack ---------------------------------------------------------------
\def\yP{-7.4}
\node[param, text width=2.6cm, minimum height=30mm] (w0) at (-6.05,\yP)
  {\textbf{Initial shares}\\ $w^0_z$\\[1pt] softmax within the patch, held before day $t_0$};
\node[param, text width=6.2cm, minimum height=30mm] (dev) at (-0.65,\yP)
  {\textbf{Deviations} $\delta_{z,t}$\\[1pt]
   AR(1) on weekly knots, summing to zero\\ in each patch,
   level scale $\sigma_L$, drift scale $\J{\sigma_{\delta,p}}$,\\
   correlation $\exp(-d_{zq}/\ell)$ with $\ell$ set by $\J{\rho_{\text{corr}}}$\\
   {\footnotesize $d_{zq}$ distance between zone centroids.}\\
   {\footnotesize Innovations only in zones with $\ge 30$ confirmed}\\
   {\footnotesize cases, in patches with two such zones.}};

\node[latent, text width=\ws] (ren) at (\xs,-10.6)
  {\textbf{Zone renewal} from day $t_0$\\[2pt]
   $\Lambda_{z,t} = \sum_{s \ge 1} \J{\bar g_s}\, I_{z,t-s}$, \qquad
   $u_{z,t} = \exp(\delta_{z,t})\, \Lambda_{z,t}$\\[2pt]
   $u$ sets each zone's share. The patch total is the sampled $\J{I_{p,t}}$.\\[1pt]
   {\footnotesize $t_0$ is 42 days before the first zone table. Before it,
   $I_{z,t} = w^0_z \J{I_{p,t}}$. Zone $R_{z,t} = I_{z,t} / \Lambda_{z,t}$.}};

\node[latent, text width=\ws] (arr) at (\xs,-14.3)
  {\textbf{(a) Arrivals from other patches: how many and where they land}\\[3pt]
   $M_{p,t} = f_{p,t}\, \J{I_{p,t}}$\\[2pt]
   $\operatorname{logit} f_{p,t} = \J{\bar o_{p,t}}
     + \log \sum_{q \ne p} \J{s_{pq,t}}\,
     \exp\bigl(\J{\zeta^{\varepsilon}_q} + \J{\zeta_q(t)}\bigr) - \J{\zeta_p(t)}$\\[3pt]
   $h_{z,t} = \sum_{z':\, p(z') \ne p(z)} \J{\varepsilon_{p(z')}}\, K^{\text{b}}_{zz'}\, u_{z',t}$\\[2pt]
   {\footnotesize $K^{\text{b}}$: the joint kernel spread over zones by gravity
   pull, so it sums to $K$ over a patch. $\J{\varepsilon_{p(z')}}$ only weights
   origins against each other. $p(z')$ is the patch of zone $z'$.}};

\node[latent, text width=\ws] (alloc) at (\xs,-18.45)
  {\textbf{(b) Allocation within the patch}\\[2pt]
   $v_{z,t} = (1 - \varepsilon_z)\, u_{z,t}
     + \sum_{z' \in p,\, z' \ne z} \varepsilon_{z'} K^{\text{w}}_{zz'}\, u_{z',t}$\\[3pt]
   $I_{z,t} = (\J{I_{p,t}} - M_{p,t})\,
     \dfrac{v_{z,t}}{\sum_{z' \in p} v_{z',t}}
     + M_{p,t}\, \dfrac{h_{z,t}}{\sum_{z' \in p} h_{z',t}}$\\[2pt]
   so $\sum_{z \in p} I_{z,t} = \J{I_{p,t}}$\\[2pt]
   {\footnotesize $K^{\text{w}}$: gravity pull among the zones of one patch,
   normalised over that patch. $\varepsilon_z$ within-patch intensity, a zone
   parameter pooled around one level $\varepsilon_{\text{w}}$.
   Without mixing, $I_{z,t} = \J{I_{p,t}}\, u_{z,t} / \sum_{z' \in p} u_{z',t}$.}};

\node[latent, text width=\ws] (exc) at (\xs,-22.25)
  {\textbf{Expected confirmed cases and deaths} in the window of table $v$\\[2pt]
   $C_{z,v} = \sum_{t \in (d_{v-1}, d_v]} (\J{\bar f} * I_z)_t$, \enspace
   $D_{z,v}$ the same through the \J{death delay}\\[2pt]
   $\pi^{C}_{z,v} = \dfrac{a_z C_{z,v}}{\sum_{z' \in p} a_{z'} C_{z',v}}$, \qquad
   $\pi^{D}_{z,v} = \dfrac{\theta_z D_{z,v}}{\sum_{z' \in p} \theta_{z'} D_{z',v}}$\\[2pt]
   {\footnotesize $d_v$ date of zone table $v$. Relative ascertainment $a_z$
   (scale $\J{\sigma_a}$) and relative lethality $\theta_z$ (scale $\sigma_\theta$)
   sum to zero on the log scale within the patch.}};

\node[data, text width=\ws] (obs) at (\xs,-25.75)
  {\textbf{Zone tables} (confirmed cases and deaths by health zone)\\[2pt]
   $y_{p,v} \sim \mathrm{DirichletMultinomial}\bigl(N_{p,v},\ \kappa_{\text{DM}}\, \pi_{p,v}\bigr)$,
   \quad $\kappa_{\text{DM}} = (1 - \J{\rho}) / \J{\rho}$\\[2pt]
   {\footnotesize $N_{p,v}$ the patch's allocated rise since table $v-1$. $\J{\rho}$ is an
   intra-class correlation, and the deaths take their own $\J{\rho_{\text{death}}}$.}};

% Zone flow.
\draw[flow] (w0.south) -- (w0.south |- ren.north);
\draw[flow] (dev.south) -- (dev.south |- ren.north);
\draw[flow] (ren) -- (arr);
\draw[flow] (arr) -- (alloc);
\draw[flow] (alloc) -- (exc);
\draw[flow] (exc) -- (obs);
% The renewal reads the zone infections back on later days.
\draw[flow, dashed] (alloc.west) -- ++(-0.3,0) |- (ren.west)
  node[edgelabel, pos=0.25, rotate=90] {$I_{z,t}$ on later days};

\begin{scope}[on background layer]
  \node[group=latentline, fit={(w0) (dev) (obs) (-8.05,\yP)}, inner sep=7pt] (zgrp) {};
\end{scope}
\node[grouplabel=latentline] at (zgrp.north west) {Zones $z$ of each patch $p$};

% ---- Joint inputs, level with the step each feeds ---------------------------
\node[joint, text width=\wj] (j1) at (\xj,\yP)
  {\badge{prior} $\sigma_{\delta,p}$ drift scales,\\ $\rho_{\text{corr}}$ deviation correlation};
\node[joint, text width=\wj] (j2) at (\xj,-10.6)

\node[joint, text width=\wj] (j3) at (\xj,-14.3)

   \badge{mean} $\bar o_{p,t}$ log odds that an infection in $p$ was imported\\[2pt]
   \badge{mean} $s_{pq,t}$ share of $p$'s arrivals sent by $q$, from $\bar\varepsilon$,
   $K$ and $\bar I$};
\node[joint, text width=\wj] (j5) at (\xj,-22.25)
  {\badge{mean} $\bar f$ infection to confirmed case: incubation then receipt delay\\[2pt]

\node[joint, text width=\wj] (j6) at (\xj,-25.75)
  {\badge{prior} $\rho$, $\rho_{\text{death}}$\\ intra-class correlations};
