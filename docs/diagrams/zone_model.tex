% The health-zone model and what it takes from the joint fit. Purple marks
% a quantity taken from the joint fit; equation numbers are those of the
% health-zone section of the methods page.

\newcommand{\J}[1]{{\color{jointline}#1}}

\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm]

% ---- From the joint fit ----------------------------------------------------
\def\jw{3.75cm}
\node[joint, text width=\jw, anchor=north, minimum height=37mm] (ja) at (-4.3,0)
  {\textbf{Sampled (melded)}\\[2pt]
   weekly infections of each patch and origin intensities, as a normal
   fitted to the joint draws\\[3pt]
   $\eta \sim \mathrm{Normal}(0, I_d)$, \enspace $a = L\eta$\\[2pt]
   $I_{p,t} = \bar I_{p,t}\, e^{a_p(t)}$\\[1pt]
   $\varepsilon_q = \bar\varepsilon_q\, e^{c_q}$ \quad (56)};
\node[joint, text width=\jw, anchor=north, minimum height=37mm] (jb) at (0,0)
  {\textbf{Fixed at posterior means}\\[2pt]
   centres $\bar I_{p,t}$ and $\bar\varepsilon_q$\\
   generation interval $\bar g$\\
   case delay $\bar f = f_{\text{inc}} * f_{\text{rec}}$\\
   death delay $f_{\text{inc}} * f_d * f_{\text{rec}}$\\
   import log odds $\bar o_{p,t}$\\
   arrival shares $s_{pq,t}$};
\node[joint, text width=\jw, anchor=north, minimum height=37mm] (jc) at (4.3,0)
  {\textbf{Priors fitted to draws}\\[2pt]
   drift scale $\sigma_\delta$\\
   deviation correlation $\rho_{\text{corr}}$\\
   overdispersions $\rho$, $\rho_{\text{death}}$\\
   ascertainment scale $\sigma_a$\\[3pt]
   {\footnotesize log-normal for a scale,\\ beta for a correlation\\ or an overdispersion}};

\begin{scope}[on background layer]
  \node[group=jointline, fit=(ja) (jb) (jc), inner sep=8pt] (jgrp) {};
\end{scope}
\node[grouplabel=jointline] at (jgrp.north west)
  {From the joint fit: its posterior given the national and province data};

% ---- Zone model -------------------------------------------------------------
\node[param, text width=3.45cm, minimum height=16mm] (w0) at (-4.3,-6.0)
  {\textbf{Initial shares} $w_z$\\[1pt]
   softmax within the patch};
\node[param, text width=3.45cm, minimum height=16mm] (dev) at (4.3,-6.0)
  {\textbf{Deviations} $\delta_{z,t}$\\[1pt]
   sum to zero in the patch,\\ correlation $e^{-d_{zq}/\ell}$ (63),\\
   scales $\J{\sigma_\delta}$ and $\J{\rho_{\text{corr}}}$};

\node[latent, text width=11.9cm] (ren) at (0,-8.25)
  {\textbf{Zone renewal}\quad
   $\Lambda_{z,t} = \sum_{s \ge 1} \J{\bar g_s}\, I_{z,t-s}$, \quad
   $u_{z,t} = e^{\delta_{z,t}}\, \Lambda_{z,t}$ \hfill (57)\\[1pt]
   {\footnotesize from day $t_0$, before which each zone holds $I_{z,t} = w_z \J{I_{p,t}}$.
   The zone reproduction number is $R_{z,t} = I_{z,t} / \Lambda_{z,t}$ (67).}};

\node[param, text width=3.45cm, minimum height=16mm] (eps) at (-4.3,-10.35)
  {\textbf{Within-patch}\\ \textbf{intensity} $\varepsilon_z$\\[1pt]
   pooled around $\varepsilon_{\text{w}}$ (62)};

\node[latent, text width=11.9cm] (mov) at (0,-13.25)
  {\textbf{Movement and allocation within patch} $p$ \hfill (58--61)\\[3pt]
   within the patch: \enspace $v_{z,t} = (1 - \varepsilon_z) u_{z,t}
     + \sum_{q \in p,\, q \ne z} \varepsilon_q K^{\text{w}}_{zq}\, u_{q,t}$\\[2pt]
   from other patches: \enspace $h_{z,t} = \sum_{p(q) \ne p} \J{\varepsilon_{p(q)}}\,
     K^{\text{b}}_{zq}\, u_{q,t}$, \qquad
   imports $M_{p,t} = f_{p,t}\, \J{I_{p,t}}$\\[2pt]
   {\footnotesize $f_{p,t}$ follows from $\J{\bar o_{p,t}}$, $\J{s_{pq,t}}$
   and the sampled $a$ and $c$ (59)}\\[3pt]
   $I_{z,t} = (\J{I_{p,t}} - M_{p,t})\,
     \dfrac{v_{z,t}}{\sum_{z' \in p} v_{z',t}}
     + M_{p,t}\, \dfrac{h_{z,t}}{\sum_{z' \in p} h_{z',t}}$,
   \qquad so $\sum_{z \in p} I_{z,t} = \J{I_{p,t}}$};

\node[latent, text width=11.9cm] (exc) at (0,-16.75)
  {\textbf{Expected confirmed cases and deaths}\\[2pt]
   $C_{z,v} = \sum_{t \in (d_{v-1}, d_v]} (\J{\bar f} * I_z)_t$, \qquad
   $D_{z,v}$ the same through the death delay\\[2pt]
   $\pi^{C}_{z,v} = \dfrac{a_z C_{z,v}}{\sum_{z' \in p} a_{z'} C_{z',v}}$, \qquad
   $\pi^{D}_{z,v} = \dfrac{\theta_z D_{z,v}}{\sum_{z' \in p} \theta_{z'} D_{z',v}}$
   \hfill (64, 65)\\[2pt]
   {\footnotesize $a_z$ and $\theta_z$ sum to zero on the log scale within the
   patch, at scales $\J{\sigma_a}$ and $\sigma_\theta \sim \mathrm{Normal}^{+}(0, 0.1)$ (66)}};

\draw[flow] (w0.south) -- (ren.north -| w0.south);
\draw[flow] (dev.south) -- (ren.north -| dev.south);
\draw[flow] (eps.south) -- (eps.south |- mov.north);
\draw[flow] (ren) -- (mov);
\draw[flow] (mov) -- (exc);

\begin{scope}[on background layer]
  \node[group=latentline, fit=(w0) (dev) (exc), inner sep=8pt] (zgrp) {};
\end{scope}
\node[grouplabel=latentline] at (zgrp.north west)
  {Health-zone model: zones $z$ of each patch $p$};

% ---- Data -------------------------------------------------------------------
\node[data, text width=11.9cm] (obs) at (0,-20.15)
  {\textbf{Zone tables} (confirmed cases and deaths by health zone)\\[2pt]
   $y_{p,v} \sim \mathrm{DirichletMultinomial}\bigl(N_{p,v},\ \kappa\, \pi_{p,v}\bigr)$,
   \quad $\kappa = (1 - \J{\rho}) / \J{\rho}$ \hfill (64)\\[1pt]
   {\footnotesize The deaths take their own $\J{\rho_{\text{death}}}$.
   Nothing returns to the joint fit.}};
\draw[flow] (exc) -- (obs);

% ---- Joint inputs into the zone model ---------------------------------------
\draw[jointflow] (ja.west) -- (ja.west -| -6.75,0) |- (mov.west)
  node[edgelabel, pos=0.5, fill=white, text=jointline, rotate=90, anchor=south]
  {sampled $I_{p,t}$ and $\varepsilon_q$};
\draw[jointflow] (jb.south) -- (ren.north -| jb.south)
  node[edgelabel, pos=0.66, left=3pt, text=jointline, fill=none]
  {$\bar g$, $\bar f$, death delay,\\ $\bar o_{p,t}$, $s_{pq,t}$};
\draw[jointflow] (jc.south) -- (dev.north);
\node[note, anchor=north east, text=jointline] at ([yshift=-2pt]zgrp.south east)
  {Purple symbols are taken from the joint fit.};

\end{tikzpicture}
