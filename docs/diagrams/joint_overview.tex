% The joint model as a generative chain: parameters, the patch renewal,
% symptom onsets and the national observation streams. Equations are kept
% for the infection process and the suspected cases; the other streams are
% described in words.

\begin{document}
\begin{tikzpicture}[x=1cm,y=1cm]

% Latent column [-5.2, 1.0], data column [1.6, 6.6], trunks to the left.
\def\xl{-2.1}
\def\xr{4.6}
\def\wl{6.2cm}
\def\wr{5.1cm}
\def\xa{-3.7}     % left half of the latent column
\def\xb{-0.5}     % right half of the latent column
\def\wh{3.0cm}
\def\bon{-5.6}    % onsets trunk
\def\binf{-6.0}   % infections trunk
\def\bseed{-6.4}  % seeding trunk

% ---- Parameters ----------------------------------------------------------
\node[param, text width=3.4cm, minimum height=26mm] (seed) at (-4.4,0)
  {\textbf{Seeding}\\[1pt] growth rate $r$ over $m$\\ generations gives the\\
   seed $C_T$, all in Ituri};
\node[param, text width=3.4cm, minimum height=26mm] (rt) at (0.05,0)
  {\textbf{Reproduction}\\ \textbf{number}\\[1pt]
   $R_{p,t} = R^{\text{trend}}_t \exp(\delta_{p,t})$\\
   weekly trend from $R_0$,\\ patch deviation $\delta_{p,t}$};
\node[param, text width=3.4cm, minimum height=26mm] (gi) at (5.25,0)
  {\textbf{Generation}\\ \textbf{interval} $g_s$\\[1pt]
   probability of a lag\\ of $s$ days between\\ infections};
\draw[flow] (gi.west) -- (rt.east)
  node[edgelabel, midway, above=4pt, fill=none] {$R_0$ from\\ $r$ and $g_s$};

% ---- Infections ----------------------------------------------------------
\node[latent, text width=10.9cm, inner sep=7pt] (inf) at (0.75,-3.0)
  {\textbf{Infections in each patch} $p$\\[3pt]
   $G_{p,t} = R_{p,t} \sum_{s \ge 1} I_{p,t-s}\, g_s$, \qquad
   $I_{p,t} = S_{p,t-1}\bigl(1 - \exp(-Y_{p,t}/N_p)\bigr)$\\[3pt]
   {\footnotesize $G_{p,t}$ infections generated, $Y_{p,t}$ the same after importation}\\
   {\footnotesize between patches (next figure), $S_{p,t}$ susceptibles, $N_p$ population.}\\
   {\footnotesize National infections are $I_t = \sum_p I_{p,t}$.}};

\foreach \n in {seed,rt,gi}
  \draw[flow] (\n.south) -- (\n.south |- inf.north);

% ---- Onsets --------------------------------------------------------------
\node[latent, text width=10.9cm] (ons) at (0.75,-5.4)
  {\textbf{Symptom onsets}\quad
   $\text{onsets}_{t} = \sum_p (I_{p} * f_{\text{inc}})_t$\\[1pt]
   {\footnotesize $f_{\text{inc}}$ the incubation period and $*$ convolution
   with a delay distribution}};
\draw[flow] (0.75,0 |- inf.south) -- (ons.north);

% ---- Streams --------------------------------------------------------------
\def\ya{-8.2}
\def\yb{-10.6}
\def\yc{-12.65}
\def\yd{-14.6}
\def\ye{-16.55}
\def\yf{-18.45}
\def\yg{-20.3}
\def\yh{-22.1}
\def\yi{-23.9}

\node[note, anchor=south] at (\xa,{\ya+1.15}) {expected series};
\node[note, anchor=south] at (\xr,{\ya+1.15}) {observed data and likelihood};

\node[latent, text width=\wl] (rep) at (\xl,\ya)
  {\textbf{Suspected cases}\\[1pt]
   $c_t = p_{\text{DRC}}\, (\text{onsets} * f_{\text{rep}})_t + \lambda_{\text{bg},t}$\\[1pt]
   {\footnotesize $f_{\text{rep}}$ onset-to-report delay, $p_{\text{DRC}}$
   reporting fraction,}\\
   {\footnotesize $\lambda_{\text{bg},t}$ suspects who are not outbreak cases}};
\node[latent, text width=\wh, minimum height=14mm] (tc) at (\xb,\yb)
  {admissions from the\\ suspected cases, occupied\\ beds as a running\\ balance};
\node[latent, text width=\wh, minimum height=14mm] (lab) at (\xa,\yc)
  {suspected cases\\ through the receipt\\ delay, positivity};
\node[latent, text width=\wh, minimum height=14mm] (rec) at (\xb,\yd)
  {survivors ($\approx 1 - \mathrm{CFR}$) of\\ modelled confirmed\\
   cases through a\\ recovery delay};
\node[latent, text width=\wl] (cd) at (\xl,\ye)
  {analysed volume times the suspected\\ death-to-case ratio, death-pool positivity};
\node[latent, text width=\wl] (dth) at (\xl,\yf)
  {onsets through the onset-to-death delay times\\ CFR and death ascertainment, plus\\ background deaths};
\node[latent, text width=\wl] (oc) at (\xl,\yg)
  {onsets through a reporting hazard};
\node[latent, text width=\wl] (ex) at (\xl,\yh)
  {infections weighted by export propensity,\\ at risk until detected, times travel rate};
\node[latent, text width=\wl] (age) at (\xl,\yi)
  {outbreak age: cryptic phase plus\\ observed window};

\node[data, text width=\wr] (d_rep) at (\xr,\ya)
  {\textbf{Suspected cases}\\[1pt]
   $\Delta S_i \sim \mathrm{NegBin}\bigl(\textstyle\sum_{t = d_{i-1}+1}^{d_i} c_t,\ k\bigr)$\\[1pt]
   {\footnotesize $\Delta S_i$ rise between reports on days\\ $d_{i-1}$ and $d_i$, $k$ dispersion}};
\node[data, text width=\wr] (d_tc) at (\xr,\yb)
  {\textbf{Treatment centres}\\ beds, admissions, deaths, rule-outs,\\ absconds (NegBin; occupancy censored\\ at capacity, admissions at free beds)};
\node[data, text width=\wr] (d_lab) at (\xr,\yc)
  {\textbf{Analysed specimens} (NegBin)\\ \textbf{Confirmed cases}\\ (beta-binomial on analysed count)};
\node[data, text width=\wr] (d_rec) at (\xr,\yd)
  {\textbf{Recovered among confirmed}\\ (NegBin)};
\node[data, text width=\wr] (d_cd) at (\xr,\ye)
  {\textbf{Confirmed deaths} (NegBin)};
\node[data, text width=\wr] (d_dth) at (\xr,\yf)
  {\textbf{Suspected deaths} (NegBin)};
\node[data, text width=\wr] (d_oc) at (\xr,\yg)
  {\textbf{Onset curve} (Student-$t$ increments)};
\node[data, text width=\wr] (d_ex) at (\xr,\yh)
  {\textbf{Uganda exports and deaths}\\ (Poisson)};
\node[data, text width=\wr] (d_age) at (\xr,\yi)
  {\textbf{Genetic TMRCA}\\ (censored Normal)};

\foreach \a in {rep,tc,lab,rec,cd,dth,oc,ex,age}
  \draw[flow] (\a) -- (d_\a);

% Suspected-case branch: short arrows from the suspected cases.
\draw[flow] (\xb,0 |- rep.south) -- (tc.north);
\draw[flow] (\xa,0 |- rep.south) -- (lab.north);
% The laboratory feeds the recoveries and the confirmed deaths.
\draw[flow] ([xshift=0.8cm]lab.south) -- ([xshift=0.8cm]lab.south |- rec.west) -- (rec.west);
\draw[flow] ([xshift=-0.7cm]lab.south) -- ([xshift=-0.7cm]lab.south |- cd.north);
\draw[flow] (\xb,0 |- dth.north) -- (\xb,0 |- cd.south);

% Onsets: straight to the suspected cases, by a trunk to deaths and curve.
\draw[flow] (\xl,0 |- ons.south) -- (rep.north);
\draw[line width=0.9pt, draw=black!75] (ons.west) -- (\bon,0 |- ons.west) -- (\bon,\yg);
\draw[flow] (\bon,\yf) -- (dth.west);
\draw[flow] (\bon,\yg) -- (oc.west);
% Infections trunk to the exports, seeding trunk to the outbreak age.
\draw[line width=0.9pt, draw=black!75] (inf.west) -- (\binf,0 |- inf.west) -- (\binf,\yh);
\draw[flow] (\binf,\yh) -- (ex.west);
\draw[line width=0.9pt, draw=black!75] (seed.west) -- (\bseed,0 |- seed.west) -- (\bseed,\yi);
\draw[flow] (\bseed,\yi) -- (age.west);

% Trunk labels at their origins.
\node[edgelabel, rotate=90] at (\bon,{\ya-0.6}) {onsets};
\node[edgelabel, rotate=90] at (\binf,{\ya-0.6}) {infections};
\node[edgelabel, rotate=90] at (\bseed,{\ya-0.6}) {seeding};

% ---- Key -------------------------------------------------------------------
\colourkey{(-5.2,{\yi-1.2})}

\end{tikzpicture}
\end{document}
